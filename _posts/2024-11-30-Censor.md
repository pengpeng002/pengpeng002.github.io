# Censor

frog is now a editor to censor so\-called sensitive words \(敏感词\)\.

She has a long text p

\. Her job is relatively simple \-\- just to find the first occurence of sensitive word w

and remove it\.

frog repeats over and over again\. Help her do the tedious work\.

### Input

The input consists of multiple tests\. For each test:

The first line contains 1

string w\. The second line contains 1 string p

\.

\(1≤length of w,p≤5⋅106

, w,p

consists of only lowercase letter\)

### Output

For each test, write 1

string which denotes the censored text\.

### Sample Input

```
    abc
    aaabcbc
    b
    bbb
    abc
    ab
```

### Sample Output

```
    a
    
    ab
```

```cpp
#include<bits/stdc++.h>
using namespace std;
int const maxn = 5e6 + 5;
char s1[maxn], s2[maxn], ans[maxn];
int nex[maxn], pos[maxn];
//扫描一遍母串，在扫描的时候，与子串进行匹配，将母串字符压入栈，j记录匹配的字符个数\
当匹配字符个数等于子串长度时，将栈顶所有与子串匹配的字符弹出，从当前栈顶元素开始\
进行计算下一次匹配（前面字符已经匹配了）。母串扫完时，栈内所剩即为结果 
int main()
{
	while(scanf("%s%s", s2, s1) == 2)
	{
		int l1 = strlen(s1), l2 = strlen(s2);
		int i = 0, j = -1;
		nex[0] = -1;//get_next
		while(i < l1)
		{
			if(j == -1 || s2[i] == s2[j]) i++, j++, nex[i] = j;
			else j = nex[j];
		}
		i = 0, j = 0;//i指向在母串的位置，j指向在子串的位置 
		int cnt = 0;// 
		while (i < l1)//KMP匹配 
		{
			ans[cnt] = s1[i++];
			while (!(j == -1 || ans[cnt] == s2[j])) j = nex[j];//查找匹配的字符\
			直到找到了匹配的字符或j已经回溯到了-1退出 
			j++, cnt++;
			pos[cnt] = j;//记录j的位置 
			if (j == l2) cnt -= l2, j = pos[cnt];
			//数组模拟栈，当匹配字符个数与子串相等时，将栈顶所有匹配字符弹出\
			重新从栈顶元素的匹配个数开始计算下一次匹配 
		}
		for(i = 0; i < cnt; i++) printf("%c", ans[i]);
		printf("\n");
	}
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef unsigned long long ull;
const int maxn = 5e6+6, seed = 1e9+13;
char s1[maxn], s2[maxn], ans[maxn];
ull hash_s2[maxn], init_hash[maxn] = {1};
int main()
{
	for (int i = 1; i < maxn; i++) init_hash[i] = init_hash[i-1] * seed;
	while (scanf ("%s%s", s1, s2) == 2)
	{
		hash_s2[0] = 0;
		ull hash_s1 = 0, hash_top = 0;
		int len1 = strlen(s1), len2 = strlen(s2);
		for (int i = 0; i < len1; i++) hash_s1 = hash_s1 * seed + (s1[i] - 'a') + 1;
		for (int i = 0; i < len2; i++)
		{
			ans[hash_top++] = s2[i];
			hash_s2[hash_top] = hash_s2[hash_top-1] * seed + (s2[i] - 'a') + 1;
			if (hash_top >= len1 && hash_s2[hash_top] - hash_s2[hash_top - len1] * init_hash[len1] == hash_s1)
				hash_top -= len1;
		}
		for (int i = 0; i < hash_top; i++)printf("%c", ans[i]);
		puts("");
	}
}
```
